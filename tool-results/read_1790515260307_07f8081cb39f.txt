   170→def _fold(s: str) -> str:
   171→    t = (s or "").replace("İ", "i").replace("I", "i").replace("ı", "i").lower()
   172→    return (
   173→        t.replace("ğ", "g").replace("ş", "s").replace("ö", "o")
   174→        .replace("ü", "u").replace("ç", "c")
   175→    )

   196→def _heading_pat(label: str) -> str:
   197→    parts: list[str] = []
   198→    for ch in label:
   199→        if ch in "İIıi":
   200→            parts.append("[İIıi]")
   201→        elif ch in "Şş":
   202→            parts.append("[ŞşSs]")
   203→        elif ch in "Çç":
   204→            parts.append("[ÇçCc]")
   205→        elif ch in "Ğğ":
   206→            parts.append("[ĞğGg]")
   207→        elif ch in "Üü":
   208→            parts.append("[ÜüUu]")
   209→        elif ch in "Öö":
   210→            parts.append("[ÖöOo]")
   211→        elif ch in r"\^$.|?*+()[]{}":
   212→            parts.append("\\" + ch)
   213→        else:
   214→            parts.append(ch)
   215→    return "".join(parts)

   218→def _section(text: str, start: str, end: str = "") -> str:
   219→    t = text or ""
   220→    m = re.search(rf"(?im)^{_heading_pat(start)}\s*:?\s*$", t)
   221→    if not m:
   222→        return ""
   223→    rest = t[m.end() :]
   224→    if end:
   225→        n = re.search(rf"(?im)^{_heading_pat(end)}\s*:?\s*$", rest)
   226→        if n:
   227→            return rest[: n.start()].strip()
   228→    return rest.strip()

   252→def _nearest_amount_label(text: str, start: int, end: int) -> str:
   253→    line_start = (text or "").rfind("\n", 0, start) + 1
   254→    line_end = (text or "").find("\n", start)
   255→    if line_end < 0:
   256→        line_end = len(text or "")
   257→    line = (text or "")[line_start:line_end]
   258→    loc_s = start - line_start
   259→    loc_e = max(loc_s, end - line_start)
   260→    after = line[loc_e:]
   261→    nxt = _TL_RE.search(after)
   262→    chunk = after[: nxt.start()] if nxt else after
   263→    for lab, pat in _AMOUNT_LABELS:
   264→        if re.search(pat, chunk, re.I):
   265→            return lab
   266→    before = line[:loc_s]
   267→    for lab, pat in reversed(list(_AMOUNT_LABELS)):
   268→        if re.search(pat, before, re.I):
   269→            return lab
   270→    return "diger"

   273→def _amount_hits(text: str, section: str) -> list[dict]:
   274→    out: list[dict] = []
   275→    for m in _TL_RE.finditer(text or ""):
   276→        val = _tl(m.group(1))
   277→        if val is None or val < 50:
   278→            continue
   279→        out.append(
   280→            {
   281→                "label": _nearest_amount_label(text, m.start(), m.end()),
   282→                "value": val,
   283→                "section": section,
   284→                "raw": m.group(0),
   285→            }
   286→        )
   287→    return out

   290→def _people(text: str) -> list[str]:
   291→    found: list[str] = []
   292→    extra = re.compile(
   293→        r"(?i)m[uü]vekkil(?:im)?\s+([A-ZÇĞİÖŞÜ][A-Za-zÇĞİÖŞÜçğıöşü]+(?:\s+[A-ZÇĞİÖŞÜ][A-Za-zÇĞİÖŞÜçğıöşü]+){1,2})"
   294→    )
   295→    for rx in (extra, _NAME_UP_RE, _NAME_RE):
   296→        for m in rx.finditer(text or ""):
   297→            name = re.sub(r"\s+", " ", m.group(1)).strip()
   298→            low = _fold(name)
   299→            if any(w in _STOP for w in low.split()):
   300→                continue
   301→            if len(name.split()) < 2:
   302→                continue
   303→            if name not in found:
   304→                found.append(name)
   305→    return found

   308→def _line_val(text: str, *labels: str) -> str:
   309→    for lab in labels:
   310→        m = re.search(rf"(?im)^{re.escape(lab)}\s*[:：]\s*(.+)$", text or "")
   311→        if m and m.group(1).strip():
   312→            return m.group(1).strip()
   313→    return ""

   323→def extract_model(form: dict, text: str = "") -> dict[str, Any]:
   324→    """Yapılandırılmış dilekçe kartı. Eksik alanı uydurmaz."""
   325→    kind = resolve_kind(str(form.get("petitionType") or ""))
   326→    src = _blob(form)
   327→    t = text or ""
   328→    davaci = _line_val(src, "DAVACI", "Davacı") or _header_name(t, "DAVACI")
   329→    davali = _line_val(src, "DAVALI", "Davalı") or _header_name(t, "DAVALI")
   330→    vekil = str(form.get("lawyerName") or "") or _line_val(src, "VEKİL", "Vekil")
   331→    fold = _fold(src)
   332→    ground = ""
   333→    if re.search(r"tmk\s*(?:m\.?\s*)?164|terke dayali|terk davasi", fold):
   334→        ground = "164"
   335→    elif re.search(r"tmk\s*(?:m\.?\s*)?166|temelinden sars|siddetli gecimsiz", fold):
   336→        ground = "166"
   337→    evlilik = ""
   338→    em = re.search(
   339→        r"(?i)(?:evl[ie]n|nik[aâ]h|evlilik).{0,24}(\d{1,2}[./]\d{1,2}[./]\d{4})",
   340→        src + "\n" + t,
   341→    )
   342→    if em:
   343→        evlilik = em.group(1)
   344→    children = []
   345→    for m in re.finditer(
   346→        r"(?i)(?:müşterek çocuk|çocuk(?:ları)?|velayet)\s*[:：]?\s*([A-ZÇĞİÖŞÜ][A-Za-zÇĞİÖŞÜçğıöşü]+(?:\s+[A-ZÇĞİÖŞÜ][A-Za-zÇĞİÖŞÜçğıöşü]+){1,2})",
   347→        src + "\n" + t,
   348→    ):
   349→        children.append(re.sub(r"\s+", " ", m.group(1)).strip())
   350→    acik = (
   351→        _section(t, "AÇIKLAMALAR", "HUKUKİ SEBEPLER")
   352→        or _section(t, "AÇIKLAMALAR", "DELİLLER")
   353→        or _section(t, "AÇIKLAMALAR", "HUKUKİ DAYANAKLAR")
   354→    )
   355→    son = _section(t, "SONUÇ VE İSTEM", "EKLER")
   356→    delil = (
   357→        _section(t, "DELİLLER", "SONUÇ VE İSTEM")
   358→        or _section(t, "DELİLLER", "HUKUKİ DAYANAKLAR")
   359→        or _section(t, "DELİLLER", "HUKUKİ NEDENLER")
   360→        or _section(t, "DELİLLER", "HUKUKİ SEBEPLER")
   361→    )
   362→    huk = (
   363→        _section(t, "HUKUKİ SEBEPLER", "DELİLLER")
   364→        or _section(t, "HUKUKİ SEBEPLER", "SONUÇ VE İSTEM")
   365→        or _section(t, "HUKUKİ DAYANAKLAR", "SONUÇ VE İSTEM")
   366→        or _section(t, "HUKUKİ NEDENLER", "SONUÇ VE İSTEM")
   367→    )
   368→    amounts = (
   369→        _amount_hits(src, "form")
   370→        + _amount_hits(acik, "aciklama")
   371→        + _amount_hits(son, "sonuc")
   372→        + _amount_hits(_section(t, "KONU", "AÇIKLAMALAR"), "konu")
   373→    )
   374→    return {
   375→        "kind": kind,
   376→        "dava_turu": str(form.get("petitionType") or ""),
   377→        "mahkeme": str(form.get("court") or ""),
   378→        "davaci": davaci,
   379→        "davali": davali,
   380→        "vekil": vekil,
   381→        "cocuklar": children,
   382→        "evlilik": evlilik,
   383→        "bosanma_sebebi": ground,
   384→        "olaylar": (str(form.get("caseSummary") or "") or acik)[:800],
   385→        "talepler": str(form.get("requests") or ""),
   386→        "deliller": delil,
   387→        "hukuki": huk,
   388→        "sonuc": son or str(form.get("requests") or ""),
   389→        "ekler": _section(t, "EKLER"),
   390→        "amounts": amounts,
   391→        "people_header": {
   392→            "davaci": [davaci] if davaci else [],
   393→            "davali": [davali] if davali else [],
   394→        },
   395→        "people_body": _people(acik),
   396→        "people_sonuc": _people(son),
   397→        "cites_text": [m.group(0).strip() for m in _CITE_EK_RE.finditer(t)]
   398→        + [m.group(0).strip() for m in _CITE_BNO_RE.finditer(t)],
   399→        "source": src,
   400→        "has_text": bool(t.strip()),
   401→    }

   408→def _has_field(model: dict, key: str) -> bool:
   409→    t = (model.get("source") or "") + "\n" + (model.get("olaylar") or "") + "\n" + (model.get("sonuc") or "")
   410→    fold = _fold(t + "\n" + str(model.get("mahkeme") or ""))
   411→    checks = {
   412→        "mahkeme": bool(model.get("mahkeme")) or "mahkeme" in fold,
   413→        "davaci": bool(model.get("davaci")),
   414→        "davali": bool(model.get("davali")),
   415→        "vekil": bool(model.get("vekil")) or "av." in fold,
   416→        "evlilik": bool(model.get("evlilik")) or "evlilik" in fold or "nikah" in fold,
   417→        "bosanma_sebebi": bool(model.get("bosanma_sebebi")),
   418→        "olaylar": len(str(model.get("olaylar") or "")) > 40,
   419→        "deliller": bool(model.get("deliller")) or "delil" in fold,
   420→        "hukuki": bool(model.get("hukuki")) or "tmk" in fold or "iyuk" in fold or "hmk" in fold,
   421→        "sonuc": bool(str(model.get("sonuc") or "").strip()) or bool(model.get("talepler")),
   422→        "ise_giris": bool(re.search(r"ise giris|ise basl|işe giriş|işe başl", fold)),
   423→        "isten_cikis": bool(re.search(r"isten cik|fesih|işten ayr|isten ayr", fold)),
   424→        "ucret": bool(re.search(r"ucret|ücret|asgari|net ", fold)) or any(
   425→            a["label"] == "ucret" for a in model.get("amounts") or []
   426→        ),
   427→        "alacak": bool(re.search(r"kidem|ihbar|fazla mesai|yillik izin|alacak", fold)),
   428→        "arabuluculuk": "arabuluculuk" in fold,
   429→        "urun": bool(re.search(r"urun|ürün|hizmet|mal ", fold)),
   430→        "sozlesme": "sozlesme" in fold or "sözleşme" in fold,
   431→        "bedel": bool(model.get("amounts")) or "bedel" in fold,
   432→        "ayip": bool(re.search(r"ayip|ayıp|aykiri", fold)),
   433→        "islem": bool(re.search(r"idari islem|atama|islem|iptal", fold)),
   434→        "takip": bool(re.search(r"icra|takip|haciz", fold)),
   435→    }
   436→    return bool(checks.get(key))

   439→def _party_findings(model: dict, text: str) -> list[dict]:
   440→    out: list[dict] = []
   441→    davaci = _norm_name(model.get("davaci") or "")
   442→    acik_people = [_norm_name(x) for x in model.get("people_body") or []]

   865→def _subscores(findings: list[dict], model: dict, text: str) -> dict[str, int]:
   866→    def hit(*codes: str, sev: str | None = None) -> int:
   867→        n = 0
   868→        for f in findings:
   869→            if f["code"] in codes and (sev is None or f["sev"] == sev):
   870→                n += 1
   871→        return n
   872→
   873→    bicim = 100
   874→    if not model.get("has_text"):
   875→        bicim = 40
   876→    if not re.search(r"(?im)^DAVACI\b", text or ""):
   877→        bicim -= 20
   878→    if not re.search(r"(?im)^SONUÇ VE İSTEM\b", text or ""):
   879→        bicim -= 15
   880→    if any(f["code"] == "bicim" for f in findings):
   881→        bicim -= 10
   882→    veri = 100 - 22 * hit("taraf", "cocuk", "tutar", sev=KRITIK) - 12 * hit("taraf", "cocuk", "tutar")
   883→    hukuk = (
   884→        100
   885→        - 20 * hit("madde", "hagb", "mevzuat_graf", "dayanak_kalem", sev=KRITIK)
   886→        - 10 * hit("madde", "hagb", "mevzuat_graf", "dayanak_kalem")
   887→    )
   888→    talep = (
   889→        100
   890→        - 25 * hit("talep_capraz", "tutar", "konu_sonuc", "talep_tur", sev=KRITIK)

   912→def _overall(sub: dict[str, int]) -> int:

   945→def review(form: dict, text: str = "", memory: dict | None = None) -> dict:
   946→    """İki mod: dilekçe temiz kalır; bu rapor araştırma kutusuna gider."""
   947→    memory = _load_memory(memory)
   948→    model = extract_model(form, text)
   949→    findings = []
   950→    findings.extend(_required_findings(model))
   951→    if model.get("has_text"):
   952→        findings.extend(_party_findings(model, text))
   953→        findings.extend(_amount_findings(model))
   954→        findings.extend(_law_findings(model, text))
   955→        findings.extend(_claim_alignment_findings(model, text, form))
   956→        findings.extend(_dayanak_statute_findings(model, text, form))
   957→        findings.extend(_evidence_findings(model, text))
   958→        findings.extend(_invent_findings(model, text))
   959→        findings.extend(_cite_findings(model, memory))
   960→        findings.extend(_cite_claim_findings(model, text, form))
   961→    motor = _motor_analyze(form, text or "", memory=memory, extra=findings)
   962→    uniq = motor.get("findings") or findings
   963→    if motor.get("subscores"):
   964→        sub = motor["subscores"]
   965→        score = int(motor.get("score") or 0)
   966→    else:
   967→        sub = _subscores(uniq, model, text)
   968→        score = _overall(sub)
   969→    gaps = [f["msg"] for f in uniq if f["code"] in ("zorunlu", "bicim", "delil", "zincir")]
   970→    inc = [f["msg"] for f in uniq if f["code"] not in ("zorunlu", "bicim")]
   971→    return {
   972→        "score": score,
   973→        "breakdown": motor.get("breakdown") or {k: f"{v}/100" for k, v in sub.items()},
   974→        "subscores": sub,
   975→        "gaps": gaps,
   976→        "inconsistencies": inc,
   977→        "findings": uniq,
   978→        "model": model,
   979→        "classifier": motor.get("classifier"),
   980→        "questions": motor.get("questions") or [],
   981→        "adversarial": motor.get("adversarial") or [],
   982→        "claims": motor.get("claims") or {},
   983→        "critical": bool(motor.get("critical")),
   984→        "ok": score >= 80 and not any(f["sev"] == KRITIK for f in uniq),
   985→    }

   988→def format_report(rev: dict) -> str:
   989→    if rev.get("classifier") is not None or rev.get("adversarial") is not None:
   990→        try:
   991→            return _motor_report(rev)
   992→        except Exception:
   993→            pass
   994→    score = int(rev.get("score") or 0)
   995→    lines = [f"Dilekçe kalite testi: {score}/100"]
   996→    br = rev.get("breakdown") or {}
   997→    if br:
   998→        lines.append(" · ".join(f"{k} {v}" for k, v in br.items()))
   999→    by: dict[str, list[dict]] = {KRITIK: [], YUKSEK: [], ORTA: [], DUSUK: []}
  1000→    for f in rev.get("findings") or []:
  1001→        by.setdefault(f.get("sev") or ORTA, []).append(f)
  1002→    titles = (
  1003→        (KRITIK, "Kritik — mahkemeye sunulmadan düzeltilmeli"),
  1004→        (YUKSEK, "Yüksek"),
  1005→        (ORTA, "Orta"),
  1006→        (DUSUK, "Düşük"),
  1007→    )
  1008→    any_bad = False
  1009→    for sev, title in titles:
  1010→        items = by.get(sev) or []
  1011→        if not items:
  1012→            continue
  1013→        any_bad = True
  1014→        lines.append(f"{_SEV_MARK[sev]} {title}:")
  1015→        for f in items[:8]:
  1016→            lines.append(f"- {f['msg']}")
  1017→            if f.get("hint"):
  1018→                lines.append(f"  ({f['hint']})")
  1019→    if not any_bad:
  1020→        lines.append("YESIL Sorun bulunmayanlar: zorunlu alan, taraf, tutar, madde, içtihat.")
  1021→    elif score >= 80 and not by.get(KRITIK):
  1022→        lines.append("Kritik yok; yüksek/orta notlar araştırma kutusunda.")
  1023→    return "\n".join(lines)
