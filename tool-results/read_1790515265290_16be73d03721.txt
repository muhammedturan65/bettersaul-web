   196→   196→def _heading_pat(label: str) -> str:
   197→   197→    parts: list[str] = []
   198→   198→    for ch in label:
   199→   199→        if ch in "İIıi":
   200→   200→            parts.append("[İIıi]")
   201→   201→        elif ch in "Şş":
   202→   202→            parts.append("[ŞşSs]")
   203→   203→        elif ch in "Çç":
   204→   204→            parts.append("[ÇçCc]")
   205→   205→        elif ch in "Ğğ":
   206→   206→            parts.append("[ĞğGg]")
   207→   207→        elif ch in "Üü":
   208→   208→            parts.append("[ÜüUu]")
   209→   209→        elif ch in "Öö":
   210→   210→            parts.append("[ÖöOo]")
   211→   211→        elif ch in r"\^$.|?*+()[]{}":
   212→   212→            parts.append("\\" + ch)
   213→   213→        else:
   214→   214→            parts.append(ch)
   215→   215→    return "".join(parts)

   218→   218→def _section(text: str, start: str, end: str = "") -> str:
   219→   219→    t = text or ""
   220→   220→    m = re.search(rf"(?im)^{_heading_pat(start)}\s*:?\s*$", t)
   221→   221→    if not m:
   222→   222→        return ""
   223→   223→    rest = t[m.end() :]
   224→   224→    if end:
   225→   225→        n = re.search(rf"(?im)^{_heading_pat(end)}\s*:?\s*$", rest)
   226→   226→        if n:
   227→   227→            return rest[: n.start()].strip()
   228→   228→    return rest.strip()

   252→   252→def _nearest_amount_label(text: str, start: int, end: int) -> str:
   253→   253→    line_start = (text or "").rfind("\n", 0, start) + 1
   254→   254→    line_end = (text or "").find("\n", start)
   255→   255→    if line_end < 0:
   256→   256→        line_end = len(text or "")
   257→   257→    line = (text or "")[line_start:line_end]
   258→   258→    loc_s = start - line_start
   259→   259→    loc_e = max(loc_s, end - line_start)
   260→   260→    after = line[loc_e:]
   261→   261→    nxt = _TL_RE.search(after)
   262→   262→    chunk = after[: nxt.start()] if nxt else after
   263→   263→    for lab, pat in _AMOUNT_LABELS:
   264→   264→        if re.search(pat, chunk, re.I):
   265→   265→            return lab
   266→   266→    before = line[:loc_s]
   267→   267→    for lab, pat in reversed(list(_AMOUNT_LABELS)):
   268→   268→        if re.search(pat, before, re.I):
   269→   269→            return lab
   270→   270→    return "diger"

   285→   285→            }
   286→   286→        )

   290→   290→def _people(text: str) -> list[str]:
   291→   291→    found: list[str] = []
   292→   292→    extra = re.compile(
   293→   293→        r"(?i)m[uü]vekkil(?:im)?\s+([A-ZÇĞİÖŞÜ][A-Za-zÇĞİÖŞÜçğıöşü]+(?:\s+[A-ZÇĞİÖŞÜ][A-Za-zÇĞİÖŞÜçğıöşü]+){1,2})"
   294→   294→    )

   296→   296→        for m in rx.finditer(text or ""):
   297→   297→            name = re.sub(r"\s+", " ", m.group(1)).strip()
   298→   298→            low = _fold(name)
   299→   299→            if any(w in _STOP for w in low.split()):

   302→   302→                continue
   303→   303→            if name not in found:
   304→   304→                found.append(name)
   305→   305→    return found

   357→   357→        _section(t, "DELİLLER", "SONUÇ VE İSTEM")

   361→   361→    )

   363→   363→        _section(t, "HUKUKİ SEBEPLER", "DELİLLER")
   364→   364→        or _section(t, "HUKUKİ SEBEPLER", "SONUÇ VE İSTEM")
   365→   365→        or _section(t, "HUKUKİ DAYANAKLAR", "SONUÇ VE İSTEM")
   366→   366→        or _section(t, "HUKUKİ NEDENLER", "SONUÇ VE İSTEM")
   367→   367→    )
   368→   368→    amounts = (
   369→   369→        _amount_hits(src, "form")
   370→   370→        + _amount_hits(acik, "aciklama")
   371→   371→        + _amount_hits(son, "sonuc")
   372→   372→        + _amount_hits(_section(t, "KONU", "AÇIKLAMALAR"), "konu")
   373→   373→    )
   374→   374→    return {

   376→   376→        "dava_turu": str(form.get("petitionType") or ""),

   379→   379→        "davali": davali,
   380→   380→        "vekil": vekil,
   381→   381→        "cocuklar": children,
   382→   382→        "evlilik": evlilik,
   383→   383→        "bosanma_sebebi": ground,
   384→   384→        "olaylar": (str(form.get("caseSummary") or "") or acik)[:800],
   385→   385→        "talepler": str(form.get("requests") or ""),
   386→   386→        "deliller": delil,
   387→   387→        "hukuki": huk,
   388→   388→        "sonuc": son or str(form.get("requests") or ""),
   389→   389→        "ekler": _section(t, "EKLER"),
   390→   390→        "amounts": amounts,
   391→   391→        "people_header": {
   392→   392→            "davaci": [davaci] if davaci else [],
   393→   393→            "davali": [davali] if davali else [],

   865→   865→def _subscores(findings: list[dict], model: dict, text: str) -> dict[str, int]:
   866→   866→    def hit(*codes: str, sev: str | None = None) -> int:
   867→   867→        n = 0
   868→   868→        for f in findings:
   869→   869→            if f["code"] in codes and (sev is None or f["sev"] == sev):
   870→   870→                n += 1
   871→   871→        return n
   872→   872→
   873→   873→    bicim = 100
   874→   874→    if not model.get("has_text"):
   875→   875→        bicim = 40
   876→   876→    if not re.search(r"(?im)^DAVACI\b", text or ""):
   877→   877→        bicim -= 20
   878→   878→    if not re.search(r"(?im)^SONUÇ VE İSTEM\b", text or ""):
   879→   879→        bicim -= 15
   880→   880→    if any(f["code"] == "bicim" for f in findings):
   881→   881→        bicim -= 10
   882→   882→    veri = 100 - 22 * hit("taraf", "cocuk", "tutar", sev=KRITIK) - 12 * hit("taraf", "cocuk", "tutar")
   883→   883→    hukuk = (
   884→   884→        100
   885→   885→        - 20 * hit("madde", "hagb", "mevzuat_graf", "dayanak_kalem", sev=KRITIK)
   886→   886→        - 10 * hit("madde", "hagb", "mevzuat_graf", "dayanak_kalem")
   887→   887→    )
   888→   888→    talep = (
   889→   889→        100
   890→   890→        - 25 * hit("talep_capraz", "tutar", "konu_sonuc", "talep_tur", sev=KRITIK)
   891→   891→        - 10 * hit("talep_capraz", "konu_sonuc", "talep_tur")
   892→   892→    )
   893→   893→    delil = 100 - 12 * hit("delil")
   894→   894→    dil = 90 if re.search(r"(?i)müvekkil", text or "") else 70
   895→   895→    if re.search(r"(?i)wikipedia|Vekile Not|formda 164|4\.\s*BÖLÜM", text or ""):
   896→   896→        dil -= 25
   897→   897→    ictihat = 100 - 18 * hit("ictihat")
   898→   898→    if model.get("has_text") and not (model.get("cites_text") or []):
   899→   899→        ictihat = 70
   900→   900→    clip = lambda n: max(0, min(100, n))
   901→   901→    return {
   902→   902→        "Biçimsel uygunluk": clip(bicim),
   903→   903→        "Veri tutarlılığı": clip(veri),
   904→   904→        "Hukuki dayanak": clip(hukuk),
   905→   905→        "Talep tutarlılığı": clip(talep),
   906→   906→        "Delil yeterliliği": clip(delil),
   907→   907→        "Dil ve üslup": clip(dil),
   908→   908→        "İçtihat doğrulama": clip(ictihat),
   909→   909→    }

   912→   912→def _overall(sub: dict[str, int]) -> int:
   913→   913→    w = {
   914→   914→        "Biçimsel uygunluk": 10,
   915→   915→        "Veri tutarlılığı": 25,
   916→   916→        "Hukuki dayanak": 15,

   920→   920→        "İçtihat doğrulama": 15,
   921→   921→    }

   945→   945→def review(form: dict, text: str = "", memory: dict | None = None) -> dict:
   946→   946→    """İki mod: dilekçe temiz kalır; bu rapor araştırma kutusuna gider."""
   947→   947→    memory = _load_memory(memory)
   948→   948→    model = extract_model(form, text)
   949→   949→    findings = []
   950→   950→    findings.extend(_required_findings(model))
   951→   951→    if model.get("has_text"):
   952→   952→        findings.extend(_party_findings(model, text))
   953→   953→        findings.extend(_amount_findings(model))
   954→   954→        findings.extend(_law_findings(model, text))
   955→   955→        findings.extend(_claim_alignment_findings(model, text, form))
   956→   956→        findings.extend(_dayanak_statute_findings(model, text, form))

   959→   959→        findings.extend(_cite_findings(model, memory))
   960→   960→        findings.extend(_cite_claim_findings(model, text, form))
   961→   961→    motor = _motor_analyze(form, text or "", memory=memory, extra=findings)
   962→   962→    uniq = motor.get("findings") or findings
   963→   963→    if motor.get("subscores"):
   964→   964→        sub = motor["subscores"]
   965→   965→        score = int(motor.get("score") or 0)
   966→   966→    else:
   967→   967→        sub = _subscores(uniq, model, text)
   968→   968→        score = _overall(sub)
   969→   969→    gaps = [f["msg"] for f in uniq if f["code"] in ("zorunlu", "bicim", "delil", "zincir")]
   970→   970→    inc = [f["msg"] for f in uniq if f["code"] not in ("zorunlu", "bicim")]
   971→   971→    return {
   972→   972→        "score": score,
   973→   973→        "breakdown": motor.get("breakdown") or {k: f"{v}/100" for k, v in sub.items()},
   974→   974→        "subscores": sub,
   975→   975→        "gaps": gaps,
   976→   976→        "inconsistencies": inc,
   977→   977→        "findings": uniq,
   978→   978→        "model": model,
   979→   979→        "classifier": motor.get("classifier"),
   980→   980→        "questions": motor.get("questions") or [],
   981→   981→        "adversarial": motor.get("adversarial") or [],
   982→   982→        "claims": motor.get("claims") or {},
   983→   983→        "critical": bool(motor.get("critical")),
   984→   984→        "ok": score >= 80 and not any(f["sev"] == KRITIK for f in uniq),
   985→   985→    }
