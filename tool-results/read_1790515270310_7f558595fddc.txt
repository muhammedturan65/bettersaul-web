   106→   106→def _kind_from_ptype(ptype: str) -> str:
   107→   107→    p = _fold(ptype)
   108→   108→    for code, spec in TRACKS.items():
   109→   109→        if spec.get("kind") and spec["kind"] != "diger" and spec["kind"] in p:
   110→   110→            pass
   111→   111→    if "idar" in p or "vergi" in p:
   112→   112→        return "idare"
   113→   113→    if "bosan" in p or "aile" in p:
   114→   114→        return "bosanma"
   115→   115→    if re.search(r"\bis\b|isci|işçi", p) or "iş " in (ptype or "").lower():
   116→   116→        return "is"
   117→   117→    if "icra" in p:
   118→   118→        return "icra"
   119→   119→    if "tuketici" in p:
   120→   120→        return "tuketici"
   121→   121→    if "kira" in p or "tahliye" in p:
   122→   122→        return "kira"
   123→   123→    if "tazminat" in p:
   124→   124→        return "tazminat"
   125→   125→    if "ceza" in p:
   126→   126→        return "ceza"
   127→   127→    if "alacak" in p or "ticaret" in p:
   128→   128→        return "alacak"
   129→   129→    return "diger"

   132→   132→def classify(form: dict) -> dict:
   133→   133→    """Kesin hüküm değil; ön değerlendirme."""
   134→   134→    blob = _fold(intent_text(form) + " " + str((form or {}).get("petitionType") or ""))
   135→   135→    scores: list[tuple[int, str]] = []
   136→   136→    for code, spec in TRACKS.items():
   137→   137→        n = len(re.findall(spec["pat"], blob, re.I))
   138→   138→        if n:
   139→   139→            scores.append((n, code))
   140→   140→    scores.sort(reverse=True)
   141→   141→    top = scores[0][1] if scores else _kind_from_ptype(str((form or {}).get("petitionType") or ""))
   142→   142→    if top == "diger" and str((form or {}).get("petitionType") or ""):
   143→   143→        mapped = _kind_from_ptype(str(form.get("petitionType") or ""))
   144→   144→        top = mapped if mapped != "diger" else top
   145→   145→    alts = [c for _, c in scores[1:3]]
   146→   146→    label = TRACKS.get(top, {}).get("label") or top
   147→   147→    note = f"Ön değerlendirme: Olaylarınız {label} davası niteliğinde olabilir. Kesin nitelendirme değildir."
   148→   148→    if alts:
   149→   149→        note += " Ayrım: " + " / ".join(TRACKS[a]["label"] for a in alts if a in TRACKS) + " da değerlendirilebilir."
   150→   150→    return {"track": top, "alts": alts, "note": note, "claims": sorted(form_claims(form))}

   153→   153→def _heading_pat(label: str) -> str:
   154→   154→    parts: list[str] = []
   155→   155→    for ch in label:
   156→   156→        if ch in "İIıi":
   157→   157→            parts.append("[İIıi]")
   158→   158→        elif ch in "ŞşSs":
   159→   159→            parts.append("[ŞşSs]")
   160→   160→        elif ch in "ÇçCc":
   161→   161→            parts.append("[ÇçCc]")
   162→   162→        elif ch in "ĞğGg":
   163→   163→            parts.append("[ĞğGg]")
   164→   164→        elif ch in "ÜüUu":
   165→   165→            parts.append("[ÜüUu]")
   166→   166→        elif ch in "ÖöOo":
   167→   167→            parts.append("[ÖöOo]")
   168→   168→        elif ch in r"\^$.|?*+()[]{}":
   169→   169→            parts.append("\\" + ch)
   170→   170→        else:
   171→   171→            parts.append(ch)
   172→   172→    return "".join(parts)

   175→   175→def section(text: str, start: str, end: str = "") -> str:
   176→   176→    t = text or ""
   177→   177→    m = re.search(rf"(?im)^{_heading_pat(start)}\s*:?\s*$", t)
   178→   178→    if not m:
   179→   179→        return ""
   180→   180→    rest = t[m.end() :]
   181→   181→    if end:
   182→   182→        n = re.search(rf"(?im)^{_heading_pat(end)}\s*:?\s*$", rest)
   183→   183→        if n:
   184→   184→            return rest[: n.start()].strip()
   185→   185→    return rest.strip()

   222→   222→def extract_dates(text: str) -> list[dict]:
   223→   223→    out: list[dict] = []
   224→   224→    for m in _DATE_RE.finditer(text or ""):
   225→   225→        dt = _parse_date(m)
   226→   226→        if not dt:
   227→   227→            continue
   228→   228→        start = max(0, m.start() - 40)
   229→   229→        ctx = _fold((text or "")[start : m.end() + 20])
   230→   230→        label = "diger"
   231→   231→        for key, pat in (
   232→   232→            ("fesih", r"fesih|isten cik|isten ayr"),
   233→   233→            ("arabulucu", r"arabulucu|son tutanak"),
   234→   234→            ("ihtar", r"ihtar|temerrut|ihtarnam"),
   235→   235→            ("sozlesme", r"sozlesme|ise giris|kira baslang"),
   236→   236→            ("teblig", r"teblig|ogrenme"),
   237→   237→            ("evlilik", r"evlilik|nikah"),
   238→   238→            ("islem", r"islem|atama|karar tarihi"),
   239→   239→        ):
   240→   240→            if re.search(pat, ctx):
   241→   241→                label = key
   242→   242→                break
   243→   243→        out.append({"label": label, "date": dt, "raw": m.group(0)})
   244→   244→    return out

   247→   247→def missing_questions(form: dict, track: str) -> list[str]:
   248→   248→    blob = _fold(intent_text(form))
   249→   249→    qs: list[str] = []
   250→   250→    for key in (track,):
   251→   251→        for prio, item in MISSING_MATRIX.get(key, ()):
   252→   252→            low = _fold(item)
   253→   253→            if any(w in blob for w in low.split()[:2]):
   254→   254→                continue
   255→   255→            qs.append(f"[{prio}] {item}")
   256→   256→    claims = form_claims(form)
   257→   257→    if "ise_iade" in claims:
   258→   258→        for prio, item in MISSING_MATRIX.get("ise_iade", ()):
   259→   259→            qs.append(f"[{prio}] {item}")
   260→   260→    seen: set[str] = set()
   261→   261→    uniq: list[str] = []
   262→   262→    for q in qs:
   263→   263→        if q in seen:
   264→   264→            continue
   265→   265→        seen.add(q)
   266→   266→        uniq.append(q)
   267→   267→    return uniq[:6]

   270→   270→def _alignment(form: dict, secs: dict[str, str]) -> list[dict]:
   271→   271→    out: list[dict] = []
   272→   272→    k = detect_claims(_claim_scan(secs.get("konu") or "")) & TRACK_CLAIMS
   273→   273→    a = detect_claims(_claim_scan(secs.get("acik") or "")) & TRACK_CLAIMS
   274→   274→    h = detect_claims(_claim_scan(secs.get("huk") or "")) & TRACK_CLAIMS
   275→   275→    s = detect_claims(_claim_scan(secs.get("son") or "")) & TRACK_CLAIMS
   276→   276→    d = detect_claims(_claim_scan(secs.get("delil") or "")) & TRACK_CLAIMS
   277→   277→    wanted = form_claims(form) & TRACK_CLAIMS
   278→   278→    if k and s and k != s:
   279→   279→        out.append(_f(
   280→   280→            KRITIK, "konu_sonuc",
   281→   281→            f"KONU ile SONUÇ çelişiyor: KONU=[{_fmt(k)}] SONUÇ=[{_fmt(s)}].",
   282→   282→            "HMK m. 119: konu, açıklama ve sonuç aynı talep kümesi olmalı.",
   283→   283→            "talep",
   284→   284→        ))
   285→   285→    ghost = (a - s) & TRACK_CLAIMS
   286→   286→    if ghost:
   287→   287→        out.append(_f(
   288→   288→            KRITIK, "talep_tur",
   289→   289→            "Açıklamada SONUÇ'ta olmayan talep var: " + _fmt(ghost) + ".",
   290→   290→            "Açıklamalardaki her ana talep sonuçta ayrı yazılmalı.",
   291→   291→            "talep",
   292→   292→        ))
   293→   293→    extra_son = (s - (a | k | wanted)) & TRACK_CLAIMS
   294→   294→    if extra_son:
   295→   295→        out.append(_f(
   296→   296→            KRITIK, "talep_fazla",
   297→   297→            "SONUÇ'ta açıklamada desteklenmeyen yeni talep var: " + _fmt(extra_son) + ".",
   298→   298→            "",
   299→   299→            "talep",
   300→   300→        ))
   301→   301→    if wanted and s and (wanted - s):
   302→   302→        out.append(_f(
   303→   303→            YUKSEK, "talep_tur",
   304→   304→            "Formdaki kalem SONUÇ'ta yok: " + _fmt(wanted - s) + ".",
   305→   305→            "Seçilen / anlatılan dava türü ile sonuç uyumlu olmalı.",
   306→   306→            "talep",
   307→   307→        ))
   308→   308→    leftover_h = (h - (s | wanted | k)) & (LABOR_CLAIMS | FAMILY_CLAIMS)
   309→   309→    if leftover_h:
   310→   310→        out.append(_f(
   311→   311→            KRITIK, "dayanak_kalem",
   312→   312→            "Dayanak, istenmeyen kaleme uzanıyor: " + _fmt(leftover_h) + ".",
   313→   313→            "",
   314→   314→            "dayanak",
   315→   315→        ))
   316→   316→    if a and not d and secs.get("delil"):
   317→   317→        pass
   318→   318→    for need, extra, msg in JOINDER_RULES:
   319→   319→        pool = k | a | s | wanted
   320→   320→        if (need & pool) and (extra & pool):
   321→   321→            out.append(_f(YUKSEK, "joinder", msg, "Taleplerin birlikte ileri sürülmesi usulen ayrıca kontrol edilmelidir.", "usul"))
   322→   322→    son = secs.get("son") or ""
   323→   323→    if s and not re.search(r"(?i)yargılama gider|yek[aâ]let ücret", son):
   324→   324→        out.append(_f(ORTA, "sonuc_usul", "SONUÇ'ta yargılama gideri / vekâlet ücreti yok.", "", "usul"))
   325→   325→    money = bool(_TL_RE.search(son)) or bool(s & LABOR_CLAIMS)
   326→   326→    if money and not re.search(r"(?i)faiz", son):
   327→   327→        out.append(_f(ORTA, "faiz", "Para alacağı var; faiz türü / başlangıcı SONUÇ'ta yok.", "Her kaleme aynı faiz otomatik yazılmaz.", "faiz"))
   328→   328→    if (s & {"ucret", "fazla", "kira_alacak", "para_alacak"}) and not re.search(r"(?i)belirsiz alacak|HMK\s*m\.?\s*107", son + (secs.get("huk") or "")):
   329→   329→        if re.search(r"(?i)belirsiz|şimdilik|islah", intent_text(form) + son):
   330→   330→            out.append(_f(ORTA, "belirsiz", "Belirsiz alacak ifadesi zayıf; HMK m. 107 doğrulanmalı.", "", "talep"))
   331→   331→    return out

   352→   352→def _evidence_findings(form: dict, secs: dict[str, str], text: str) -> list[dict]:
   353→   353→    out: list[dict] = []
   354→   354→    delil = (secs.get("delil") or "") + "\n" + (secs.get("ek") or "")
   355→   355→    acik = secs.get("acik") or ""
   356→   356→    blob = acik + "\n" + (text or "")
   357→   357→    for code, claim_re, ev_re, title in CLAIM_EVIDENCE:
   358→   358→        if re.search(claim_re, blob) and not re.search(ev_re, delil):
   359→   359→            out.append(_f(YUKSEK, "delil", f"{title} iddiası var; bunu destekleyen delil satırı yok.", "Delil yoksa uydurulmaz.", "delil"))
   360→   360→    if re.search(r"(?i)fazla (çalış|mesai)", blob) and re.search(r"(?i)banka|dekont", delil) and not re.search(r"(?i)puantaj|pdks|tanık", delil):
   361→   361→        out.append(_f(ORTA, "delil_analiz", "Banka dekontu ücret için anlamlı olabilir; fazla çalışma saatlerini tek başına kanıtlamaz.", "", "delil_vakia"))
   362→   362→    ek = secs.get("ek") or ""
   363→   363→    if delil and ek:
   364→   364→        d_items = [ln.strip() for ln in delil.splitlines() if len(ln.strip()) > 8]
   365→   365→        if d_items and not any(_fold(x)[:18] in _fold(ek) for x in d_items[:6]):
   366→   366→            out.append(_f(ORTA, "delil_ek", "DELİL/EK UYUMSUZLUĞU: deliller ile EKLER örtüşmeyebilir.", "", "delil"))
   367→   367→    seen_alias: set[str] = set()
   368→   368→    for key, pat in EVIDENCE_ALIASES:
   369→   369→        hits = re.findall(pat, delil, re.I)
   370→   370→        if len(hits) >= 2 and key not in seen_alias:
   371→   371→            seen_alias.add(key)
   372→   372→            out.append(_f(DUSUK, "delil_tekrar", f"Aynı belge farklı adlarla tekrar: {key}.", "", "delil"))
   373→   373→    return out

   681→   681→def analyze(form: dict, text: str = "", memory: dict | None = None, extra: list[dict] | None = None) -> dict[str, Any]:
   682→   682→    secs = split_sections(text or "")

   701→   701→    for f in findings:

   707→   707→    sub = _subscores(uniq, text or "")

   710→   710→    questions = missing_questions(form, clf["track"])
   711→   711→    labeled = {DIM_LABEL[k]: v for k, v in sub.items()}
   712→   712→    crit = [f for f in uniq if f.get("sev") == KRITIK]
   713→   713→    return {
   714→   714→        "score": score,
   715→   715→        "subscores": labeled,
   716→   716→        "breakdown": {k: f"{v}/100" for k, v in labeled.items()},
