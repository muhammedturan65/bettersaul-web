   101→def _plain(html: str) -> str:
   102→    html = re.sub(r"(?is)<(script|style).*?>.*?</\1>", " ", html or "")
   103→    html = re.sub(r"(?is)<svg\b.*?</svg>", " ", html)
   104→    html = re.sub(r"(?s)<[^>]+>", " ", html)
   105→    html = re.sub(r"&nbsp;|&amp;|&quot;|&#39;", " ", html)
   106→    text = re.sub(r"\s+", " ", html).strip()
   107→    if re.search(
   108→        r"image/svg|svg\+xml|böyle bir içerik mevcut değildir|"
   109→        r"Bilgi İşlem Genel Müdürlüğü|Uygulama içerisinde böyle bir içerik|"
   110→        r"(?:HTTP\s+)?404(?:\s+Uygulama|\s+Not Found|\s+Sayfa)|"
   111→        r"Erişim Sınırı Aşıldı",
   112→        text,
   113→        re.I,
   114→    ):
   115→        return ""
   116→    return text

   222→def _tokens(phrase: str) -> list[str]:
   223→    words = re.findall(r"[A-Za-zÇĞİÖŞÜçğıöşü0-9]{3,}", phrase or "")
   224→    out: list[str] = []
   225→    for w in words:
   226→        fold = w.replace("İ", "i").replace("I", "i").lower()
   227→        if fold in _STOP_WORDS or w in out:
   228→            continue
   229→        out.append(w)
   230→        if len(out) >= 5:
   231→            break
   232→    return out or [((phrase or "").strip()[:80] or "iptal")]

   265→def _rows_from_html(html: str) -> list[dict[str, Any]]:
   266→    text = html or ""
   267→    ids = re.findall(r"getDokuman\(\s*['\"](\d{5,})['\"]", text)
   268→    ids += re.findall(r'data-(?:id|documentid)=[\'"](\d{5,})[\'"]', text, re.I)
   269→    ids += re.findall(r'documentId["\s:=]+(\d{5,})', text)
   270→    ids += re.findall(r"[?&]id=(\d{5,})", text)
   271→    ids += re.findall(r"\bid\s*[:=]\s*['\"](\d{5,})['\"]", text)
   272→    seen: set[str] = set()
   273→    ordered: list[str] = []
   274→    for doc_id in ids:
   275→        if doc_id not in seen:
   276→            seen.add(doc_id)

   279→    blocks = re.findall(
   280→        r"(?:Danıştay|Daire|Kurul)[^<]{0,80}.*?(\d{4}/\d{1,6})[^<]{0,40}(\d{4}/\d{1,6})",
   281→        _plain(text),
   282→        re.I,
   283→    )
   284→    for i, doc_id in enumerate(ordered[:40]):
   285→        esas, karar = ("", "")
   286→        if i < len(blocks):
   287→            esas, karar = blocks[i]
   288→        rows.append(
   289→            {
   290→                "documentId": doc_id,
   291→                "birimAdi": "Danıştay" if "danistay" in text.lower() else "",
   292→                "esasNo": esas,
   293→                "kararNo": karar,
   294→                "kararTarihi": "",
   295→                "baslik": "",
   296→                "ozet": "",
   297→            }
   298→        )
   299→    return rows

   350→def _portal_search(base: str, phrase: str, page_size: int = 50, page_number: int = 1) -> dict[str, Any]:
   351→    phrase = (phrase or "").strip()
   352→    if not phrase:
   353→        return {"error": "Arama ifadesi boş."}
   354→    if _is_cooling():
   355→        return {"error": "Erişim sınırı", "phrase": phrase, "kaynak": base}
   356→    size = max(10, min(int(page_size or 10), 20))
   357→    page = max(1, min(int(page_number or 1), 4))
   358→    body = _search_body(base, phrase, page, size)
   359→    last = "Arama yanıt vermedi."
   360→    with _client(base) as http:
   361→        try:
   362→            res = http.post(base + "/aramalist", json=body)
   363→        except Exception as exc:
   364→            return {"error": str(exc), "phrase": phrase, "kaynak": base}
   365→        last = f"HTTP {res.status_code}"
   366→        if _rate_limited(res):
   367→            _mark_limited()
   368→            return {"error": "Erişim sınırı", "phrase": phrase, "kaynak": base}
   369→        if res.status_code >= 400:
   370→            return {"error": last, "phrase": phrase, "kaynak": base}
   371→        try:
   372→            data = _as_dict(res.json())
   373→            if isinstance(data.get("metadata"), dict) and data["metadata"].get("FMTY") == "ERROR":
   374→                last = data["metadata"].get("FMTE") or last
   375→                return {"error": last, "phrase": phrase, "kaynak": base}
   376→            rows, total = _rows_from(data)
   377→            if not rows and data.get("data") not in (None, {}, []):
   378→                rows = _rows_from_html(_dump(data.get("data")))
   379→                total = total or len(rows)
   380→            rows = [r for r in rows if _row_fits(base, r)]
   381→            if rows:
   382→                _remember_docs(base, phrase, rows)
   383→            return {
   384→                "kaynak": base,
   385→                "phrase": phrase,
   386→                "sayfa": page,
   387→                "toplam": total,
   388→                "decisions": rows,
   389→            }
   390→        except Exception:
   391→            rows = [r for r in _rows_from_html(res.text) if _row_fits(base, r)]
   392→            if rows:
   393→                _remember_docs(base, phrase, rows)
   394→                return {
   395→                    "kaynak": base,
   396→                    "phrase": phrase,
   397→                    "sayfa": page,
   398→                    "toplam": len(rows),
   399→                    "decisions": rows,
   400→                }
   401→    return {"error": last, "phrase": phrase, "kaynak": base}

   404→def _maybe_b64(raw: str) -> str:
   405→    t = (raw or "").strip()
   406→    if len(t) < 80 or len(t) % 4:
   407→        return ""
   408→    if not re.fullmatch(r"[A-Za-z0-9+/=\s]+", t):
   409→        return ""
   410→    try:
   411→        import base64
   412→
   413→        out = base64.b64decode(t, validate=False)
   414→        for enc in ("utf-8", "cp1254", "latin-1"):
   415→            try:
   416→                text = out.decode(enc)
   417→            except Exception:
   418→                continue
   419→            if "<" in text or len(text) > 80:
   420→                return text
   421→    except Exception:
   422→        return ""
   423→    return ""

   426→def _doc_from_payload(data: Any) -> str:
   427→    if isinstance(data, str):
   428→        plain = _plain(data)
   429→        if len(plain) >= 40:
   430→            return plain[:16000]
   431→        decoded = _maybe_b64(data)
   432→        return _plain(decoded)[:16000] if decoded else ""
   433→    if not isinstance(data, dict):
   434→        return ""
   435→    meta = data.get("metadata") if isinstance(data.get("metadata"), dict) else {}
   436→    if str(meta.get("FMTY") or "").upper() == "ERROR":
   437→        return ""
   438→    blob = data.get("data")
   439→    if blob is None:
   440→        blob = data.get("content") or data.get("dokuman") or data.get("html")
   441→    if isinstance(blob, str):
   442→        return _doc_from_payload(blob)
   443→    if isinstance(blob, dict):
   444→        for key in ("content", "html", "dokuman", "document", "text", "data"):
   445→            val = blob.get(key)
   446→            if isinstance(val, str) and val.strip():
   447→                got = _doc_from_payload(val)
   448→                if got:
   449→                    return got
   450→    return ""

   453→def _decode_doc_response(res: httpx.Response) -> str:
   454→    if res.status_code >= 400:
   455→        return ""
   456→    text = res.text or ""
   457→    ctype = (res.headers.get("content-type") or "").lower()
   458→    if "json" in ctype or text.lstrip().startswith("{") or text.lstrip().startswith("["):
   459→        try:
   460→            return _doc_from_payload(res.json())
   461→        except Exception:
   462→            return ""
   463→    plain = _plain(text)
   464→    return plain[:16000] if len(plain) > 80 else ""

   467→def _portal_document(base: str, document_id: str, aranan: str = "") -> str:
   468→    document_id = (document_id or "").strip()
   469→    if not document_id:
   470→        return "documentId boş."
   471→    if _is_cooling():
   472→        return "Erişim sınırı"
   473→    hint = aranan or (_DOC_HINTS.get(document_id) or ("", ""))[1]
   474→    if "danistay" in base:
   475→        word = hint or "iptal"
   476→        qs = httpx.QueryParams({"id": document_id, "arananKelime": word})
   477→        url = f"{base}/getDokuman?{qs}"
   478→    else:
   479→        url = f"{base}/getDokuman?id={document_id}"
   480→    with _client(base) as http:
   481→        try:
   482→            res = http.get(url)
   483→        except Exception as exc:
   484→            return str(exc)
   485→        if _rate_limited(res):
   486→            _mark_limited()
   487→            return "Erişim sınırı"
   488→        text = _decode_doc_response(res)
   489→        if text:
   490→            return text
   491→        return f"HTTP {res.status_code}"

   524→    pages: int = 2,
   525→    page_size: int = 50,
   526→    court_types: list[str] | None = None,
   527→) -> tuple[list[dict[str, Any]], list[dict[str, Any]], int]:
   528→    _apply_court_lock(court_types)
   529→    packs: list[dict[str, Any]] = []
   530→    rows: list[dict[str, Any]] = []
   531→    grand = 0
   532→    for base, label in _target_bases(phrase, court_types):
   533→        taken = 0
   534→        total = 0
   535→        err = ""
   536→        for page in range(1, max(1, pages) + 1):
   537→            data = _portal_search(base, phrase, page_size=page_size, page_number=page)
   538→            if data.get("error"):
   539→                err = str(data["error"])
   540→                if page == 1:
   541→                    packs.append({"kaynak": label, "url": base, "sayfa": page, "hata": err})
   542→                break
   543→            total = int(data.get("toplam") or total or 0)
   544→            found = list(data.get("decisions") or [])
   545→            if not found:
   546→                break
   547→            before = len(rows)
   548→            for item in found:
   549→                item["kaynak"] = label
   550→                item["sorgu"] = phrase
   551→            _merge_rows(rows, found)
   552→            taken += len(rows) - before
   553→            if len(found) < 8:
   554→                break
   555→        if taken or total:
   556→            grand += total
   557→            packs.append({"kaynak": label, "url": base, "havuz": total, "alinan": taken, "sorgu": phrase})
   558→        elif err and not any(p.get("kaynak") == label and p.get("hata") for p in packs):
   559→            packs.append({"kaynak": label, "url": base, "hata": err, "sorgu": phrase})
   560→    return packs, rows, grand

   563→def extract_holding(plain: str, limit: int = 520) -> str:
   564→    text = _plain(plain or "")
   565→    if len(text) < 40:
   566→        return text[:limit]
   567→    chunks: list[str] = []
   568→    for mark in ("SONUÇ", "HÜKÜM", "GEREKÇE", "bu nedenle", "sonuç olarak", "açıklanan nedenlerle"):
   569→        idx = text.upper().find(mark.upper()) if mark.isupper() else text.lower().find(mark.lower())
   570→        if idx >= 0:
   571→            piece = text[idx : idx + 700]
   572→            chunks.append(piece)
   573→    if not chunks:
   574→        sents = [s.strip() for s in re.split(r"(?<=[.!?])\s+", text) if len(s.strip()) > 40]
   575→        keep = [
   576→            s
   577→            for s in sents
   578→            if any(
   579→                w in s.lower()
   580→                for w in (
   581→                    "boşan",
   582→                    "kusur",
   583→                    "sadakat",
   584→                    "velayet",
   585→                    "nafaka",
   586→                    "tazminat",
   587→                    "fesih",
   588→                    "alacak",
   589→                    "iptal",
   590→                    "haksız",
   591→                )
   592→            )
   593→        ]
   594→        chunks = keep[:4] or sents[:3]
   595→    out = " ".join(chunks)
   596→    out = re.sub(r"\s+", " ", out).strip()
   597→    return out[:limit]

   695→def _parse_rg_date(raw: str) -> date | None:
   696→    t = (raw or "").strip()
   697→    if not t:
   698→        return None
   699→    for fmt in ("%Y-%m-%d", "%d.%m.%Y", "%Y%m%d", "%d/%m/%Y"):
   700→        try:
   701→            return datetime.strptime(t[:10] if fmt != "%Y%m%d" else re.sub(r"\D", "", t)[:8], fmt).date()
   702→        except ValueError:
   703→            continue
   704→    m = re.search(r"(\d{4})[-./](\d{1,2})[-./](\d{1,2})", t)
   705→    if m:
   706→        try:
   707→            return date(int(m.group(1)), int(m.group(2)), int(m.group(3)))
   708→        except ValueError:
   709→            return None
   710→    m = re.search(r"(\d{1,2})[./](\d{1,2})[./](\d{4})", t)
   711→    if m:
   712→        try:
   713→            return date(int(m.group(3)), int(m.group(2)), int(m.group(1)))
   714→        except ValueError:
   715→            return None
   716→    return None

   719→def _rg_fold(s: str) -> str:
   720→    t = (s or "").replace("İ", "i").replace("I", "i")
   721→    table = str.maketrans({"ı": "i", "İ": "i", "ö": "o", "Ö": "o", "ü": "u", "Ü": "u", "ş": "s", "Ş": "s", "ğ": "g", "Ğ": "g", "ç": "c", "Ç": "c"})
   722→    return t.translate(table).lower()

   740→def _rg_entries(html: str, page_url: str = "") -> list[dict[str, str]]:
   741→    rows: list[dict[str, str]] = []
   742→    seen: set[str] = set()
   743→    base = page_url or (RG + "/")
   744→    for href, title in re.findall(r'href=["\']([^"\']+)["\'][^>]*>([^<]{8,240})', html or "", re.I):
   745→        if href.startswith("#") or "javascript:" in href.lower():
   746→            continue
   747→        if not re.search(r"(?i)eskiler|fihrist|ilanlar|\.htm|\.pdf|\d{8}-\d+", href):
   748→            continue
   749→        url = urljoin(base, href).split("#")[0]
   750→        if url in seen:
   751→            continue
   752→        tit = _plain(title)
   753→        if len(tit) < 8 or re.search(r"(?i)pdf görüntüle|önceki sayı|sonraki sayı|mobil uygulaması", tit):
   754→            continue
   755→        seen.add(url)
   756→        rows.append({"baslik": tit[:220], "url": url})
   757→        if len(rows) >= 40:
   758→            break
   759→    if not rows and page_url:
   760→        text = _plain(html)
   761→        if text:
   762→            rows.append({"baslik": text[:180], "url": page_url})
   763→    return rows

   828→def search_resmi_gazete(query: str, date_start: str = "", date_end: str = "") -> str:
   829→    q = (query or "").strip()
   830→    if not q:
   831→        return _dump({"error": "Arama ifadesi boş.", "sonuc": "taranamadı"})
   832→    start = _parse_rg_date(date_start)
   833→    end = _parse_rg_date(date_end) or start
   834→    days: list[date] = []
   835→    if start:
   836→        cur = start
   837→        last = end or start
   838→        if last < cur:
   839→            cur, last = last, cur
   840→        while cur <= last and len(days) < 14:
   841→            days.append(cur)
   842→            cur += timedelta(days=1)
   843→    else:
   844→        today = date.today()
   845→        days = [today - timedelta(days=i) for i in range(0, 8)]
   846→    found: list[dict[str, str]] = []
   847→    last = "kayıt yok"
   848→    qlow = _rg_fold(q)
   849→    with _client(RG) as http:
   850→        http.headers["Accept"] = "text/html,application/xhtml+xml,*/*"
   851→        homes = [f"{RG}/", f"{RG}/eskiler/", f"{RG}/fihrist"]
   852→        if not start:
   853→            for url in homes:
   854→                try:
   855→                    res = http.get(url)
   856→                    if res.status_code >= 400:
   857→                        last = f"HTTP {res.status_code}"
   858→                        continue
   859→                    last = f"HTTP {res.status_code}"
   860→                    html = res.text or ""
   861→                    text = _plain(html)
   862→                    fold = _rg_fold(text)
   863→                    if qlow in fold:
   864→                        idx = fold.find(qlow)
   865→                        found.append({"baslik": text[max(0, idx - 80) : idx + 220], "url": str(res.url)})
   866→                    for row in _rg_entries(html, str(res.url)):
   867→                        if qlow in _rg_fold(row.get("baslik") or "") or qlow in _rg_fold(row.get("url") or ""):
   868→                            found.append(row)
   869→                except Exception as exc:
   870→                    last = str(exc)
   871→        for day in days:
   872→            if len(found) >= 12:
   873→                break
   874→            for url in _rg_day_urls(day):
   875→                try:
   876→                    res = http.get(url)
   877→                    if res.status_code >= 400:
   878→                        continue
   879→                    last = f"HTTP {res.status_code}"
   880→                    html = res.text or ""
   881→                    if re.search(r"(?i)böyle bir içerik mevcut değildir", html):
   882→                        continue
   883→                    text = _plain(html)
   884→                    rows = _rg_entries(html, str(res.url))
   885→                    matched = [r for r in rows if qlow in _rg_fold(r.get("baslik") or "")]
   886→                    fold = _rg_fold(text)
   887→                    if not matched and qlow in fold:
   888→                        idx = fold.find(qlow)
   889→                        matched = [{"baslik": text[max(0, idx - 80) : idx + 220], "url": str(res.url), "tarih": day.isoformat()}]
   890→                    for row in matched:
   891→                        row.setdefault("tarih", day.isoformat())
   892→                        found.append(row)
   893→                    if matched:
   894→                        break
   895→                except Exception as exc:
   896→                    last = str(exc)
   897→    uniq: list[dict[str, str]] = []
   898→    seen: set[str] = set()
   899→    for row in found:
   900→        key = (row.get("url") or "") + (row.get("baslik") or "")[:80]
   901→        if key in seen:
   902→            continue
   903→        seen.add(key)
   904→        uniq.append(row)
   905→    if not uniq:
   906→        return _dump({"kaynak": "Resmî Gazete", "query": q, "sonuc": "taranamadı", "neden": last, "kayitlar": []})
   907→    return _dump({"kaynak": "Resmî Gazete", "query": q, "kayitlar": uniq[:12]})
