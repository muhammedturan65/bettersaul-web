   350→   350→def _portal_search(base: str, phrase: str, page_size: int = 50, page_number: int = 1) -> dict[str, Any]:
   351→   351→    phrase = (phrase or "").strip()
   352→   352→    if not phrase:
   353→   353→        return {"error": "Arama ifadesi boş."}
   354→   354→    if _is_cooling():
   355→   355→        return {"error": "Erişim sınırı", "phrase": phrase, "kaynak": base}
   356→   356→    size = max(10, min(int(page_size or 10), 20))
   357→   357→    page = max(1, min(int(page_number or 1), 4))
   358→   358→    body = _search_body(base, phrase, page, size)
   359→   359→    last = "Arama yanıt vermedi."
   360→   360→    with _client(base) as http:
   361→   361→        try:
   362→   362→            res = http.post(base + "/aramalist", json=body)
   363→   363→        except Exception as exc:
   364→   364→            return {"error": str(exc), "phrase": phrase, "kaynak": base}
   365→   365→        last = f"HTTP {res.status_code}"
   366→   366→        if _rate_limited(res):
   367→   367→            _mark_limited()
   368→   368→            return {"error": "Erişim sınırı", "phrase": phrase, "kaynak": base}
   369→   369→        if res.status_code >= 400:
   370→   370→            return {"error": last, "phrase": phrase, "kaynak": base}
   371→   371→        try:
   372→   372→            data = _as_dict(res.json())
   373→   373→            if isinstance(data.get("metadata"), dict) and data["metadata"].get("FMTY") == "ERROR":
   374→   374→                last = data["metadata"].get("FMTE") or last
   375→   375→                return {"error": last, "phrase": phrase, "kaynak": base}
   376→   376→            rows, total = _rows_from(data)
   377→   377→            if not rows and data.get("data") not in (None, {}, []):
   378→   378→                rows = _rows_from_html(_dump(data.get("data")))
   379→   379→                total = total or len(rows)
   380→   380→            rows = [r for r in rows if _row_fits(base, r)]
   381→   381→            if rows:
   382→   382→                _remember_docs(base, phrase, rows)
   383→   383→            return {
   384→   384→                "kaynak": base,
   385→   385→                "phrase": phrase,
   386→   386→                "sayfa": page,
   387→   387→                "toplam": total,
   388→   388→                "decisions": rows,
   389→   389→            }
   390→   390→        except Exception:
   391→   391→            rows = [r for r in _rows_from_html(res.text) if _row_fits(base, r)]
   392→   392→            if rows:
   393→   393→                _remember_docs(base, phrase, rows)
   394→   394→                return {
   395→   395→                    "kaynak": base,
   396→   396→                    "phrase": phrase,
   397→   397→                    "sayfa": page,
   398→   398→                    "toplam": len(rows),
   399→   399→                    "decisions": rows,
   400→   400→                }
   401→   401→    return {"error": last, "phrase": phrase, "kaynak": base}

   404→   404→def _maybe_b64(raw: str) -> str:
   405→   405→    t = (raw or "").strip()
   406→   406→    if len(t) < 80 or len(t) % 4:
   407→   407→        return ""
   408→   408→    if not re.fullmatch(r"[A-Za-z0-9+/=\s]+", t):
   409→   409→        return ""
   410→   410→    try:
   411→   411→        import base64
   412→   412→
   413→   413→        out = base64.b64decode(t, validate=False)
   414→   414→        for enc in ("utf-8", "cp1254", "latin-1"):
   415→   415→            try:
   416→   416→                text = out.decode(enc)
   417→   417→            except Exception:
   418→   418→                continue
   419→   419→            if "<" in text or len(text) > 80:
   420→   420→                return text
   421→   421→    except Exception:
   422→   422→        return ""
   423→   423→    return ""

   426→   426→def _doc_from_payload(data: Any) -> str:

   432→   432→        return _plain(decoded)[:16000] if decoded else ""
   433→   433→    if not isinstance(data, dict):
   434→   434→        return ""
   435→   435→    meta = data.get("metadata") if isinstance(data.get("metadata"), dict) else {}

   522→   522→def _search_bases(
   523→   523→    phrase: str,
